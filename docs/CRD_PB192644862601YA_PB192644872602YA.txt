Implementation
buat 4 tabel baru di db target. bisa pakai query di bawah atau ada di sql/02_create_summary_tables.sql

--- query ----
CREATE TABLE IF NOT EXISTS token_summary_daily (
    account_number          VARCHAR(20)       NOT NULL,
    summary_date            DATE              NOT NULL,
    usage_amount            DECIMAL(30,6)     NOT NULL DEFAULT 0,
    topup_amount            DECIMAL(30,6)     NOT NULL DEFAULT 0,
    transaction_count       BIGINT UNSIGNED   NOT NULL DEFAULT 0,
    usage_transaction_count BIGINT UNSIGNED   NOT NULL DEFAULT 0,
    topup_transaction_count BIGINT UNSIGNED   NOT NULL DEFAULT 0,
    last_balance            DECIMAL(30,6)     NULL,
    last_balance_at         DATETIME          NULL,
    last_balance_trxid      VARCHAR(64)       NULL,
    last_transaction_at     DATETIME          NULL,
    updated_at              DATETIME(6)       NOT NULL,
    PRIMARY KEY (account_number, summary_date),
    KEY idx_summary_daily_date (summary_date)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

-- Monthly bucket: derived from token_summary_daily, never from transactions.
-- last_balance comes from the latest day that HAS a balance.
CREATE TABLE IF NOT EXISTS token_summary_monthly (
    account_number          VARCHAR(20)       NOT NULL,
    summary_month           DATE              NOT NULL,
    usage_amount            DECIMAL(30,6)     NOT NULL DEFAULT 0,
    topup_amount            DECIMAL(30,6)     NOT NULL DEFAULT 0,
    transaction_count       BIGINT UNSIGNED   NOT NULL DEFAULT 0,
    usage_transaction_count BIGINT UNSIGNED   NOT NULL DEFAULT 0,
    topup_transaction_count BIGINT UNSIGNED   NOT NULL DEFAULT 0,
    last_balance            DECIMAL(30,6)     NULL,
    last_balance_at         DATETIME          NULL,
    last_balance_trxid      VARCHAR(64)       NULL,
    last_transaction_at     DATETIME          NULL,
    updated_at              DATETIME(6)       NOT NULL,
    PRIMARY KEY (account_number, summary_month),
    KEY idx_summary_monthly_month (summary_month)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

-- Current balance per account. Guarded so a backdated row cannot win.
CREATE TABLE IF NOT EXISTS token_account_snapshot (
    account_number      VARCHAR(20)   NOT NULL,
    last_balance        DECIMAL(30,6) NULL,
    last_transaction_at DATETIME      NULL,
    last_trxid          VARCHAR(64)   NULL,
    updated_at          DATETIME(6)   NOT NULL,
    PRIMARY KEY (account_number)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

-- Processing checkpoint, one row per account job.
CREATE TABLE IF NOT EXISTS token_summary_checkpoint (
    job_key         VARCHAR(100) NOT NULL,
    source_cursor   TEXT         NULL,
    summary_as_of   DATETIME(6)  NULL,
    last_success_at DATETIME(6)  NULL,
    updated_at      DATETIME(6)  NOT NULL,
    PRIMARY KEY (job_key)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
